\subsection{Git essentials for collaboration}

\begin{frame}
  \frametitle{Why Git (and what you actually need)}
  Git is a \emph{distributed version control system}.
  You use it to:
  \begin{itemize}
    \item track changes in code,
    \item collaborate without overwriting each other,
    \item recover from mistakes (yes, including stupid ones).
  \end{itemize}

  \vspace{0.5em}
  Mental model:
  \begin{itemize}
    \item \textbf{Repository}: the project + its history.
    \item \textbf{Commit}: a snapshot with a message (“what changed and why”).
    \item \textbf{Remote}: shared version (e.g. GitLab/GitHub).
  \end{itemize}

  You do \emph{not} need:
  \begin{itemize}
    \item rebasing wizardry,
    \item cherry-picking,
    \item Git archaeology.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{The only Git workflow you need (for this course)}
  Basic collaboration pattern:
  \begin{enumerate}
    \item Clone the repository
    \item Create your own branch
    \item Make changes + commit
    \item Push branch
    \item Merge via main branch
  \end{enumerate}

  \pause
  Essential commands:
  \begin{lstlisting}[language=Python]
git clone <repo-url>
git checkout -b my-branch
git status
git add file.py
git commit -m "Clear message"
git push origin my-branch
  \end{lstlisting}

  \pause
  Rules that save friendships:
  \begin{itemize}
    \item Never work directly on \texttt{main}
    \item Commit often, commit small
    \item If something breaks: Git remembers, you don’t have to
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{What Git does NOT solve for you}
  Important realism check:
  \begin{itemize}
    \item Git does not fix bad code
    \item Git does not prevent bad ideas
    \item Git does not write commit messages for you
  \end{itemize}

  \vspace{0.5em}
  Merge conflicts are not errors — they are Git saying:
  \begin{quote}
    “Two people changed the same thing.  
    You decide.”
  \end{quote}

  If you fear conflicts, you waited too long to commit.
\end{frame}


\begin{frame}[fragile]
  \frametitle{Exercise: first collaborative Git workflow}
  \textbf{Setup (teacher):}
  \begin{itemize}
    \item Create a practice repository with:
    \begin{itemize}
      \item \lstinline|README.md|
      \item \lstinline|compute.py| (simple placeholder function)
    \end{itemize}
    \item Add students as collaborators
  \end{itemize}

  \pause
  \textbf{Student task:}
  \begin{enumerate}
    \item Clone the repository
    \item Create a branch named \lstinline|firstname-lastname|
    \item Modify \lstinline|compute.py|:
      \begin{itemize}
        \item add a small function (e.g. square, cube, norm, etc.)
        \item add your name in a comment
      \end{itemize}
    \item Commit with message:
    \begin{quote}
      \lstinline|"Add <function-name> by <name>"}
    \end{quote}
    \item Push your branch
  \end{enumerate}

  \pause
  \textbf{Bonus (controlled conflict):}
  \begin{itemize}
    \item Two students edit the same line in \lstinline|README.md|
    \item Merge branches and resolve the conflict together
  \end{itemize}
\end{frame}
